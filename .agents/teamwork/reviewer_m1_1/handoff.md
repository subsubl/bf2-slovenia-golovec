hmap = np.fromfile('/home/ator/BF2/Heightmap.raw', dtype='<u2').reshape((2049, 2049))

def sample_bilinear(x, z):
    u = np.clip((x + 2048.0) / 2.0, 0.0, 2048.0)
    v = np.clip((z + 2048.0) / 2.0, 0.0, 2048.0)
    c0, r0 = int(np.floor(u)), int(np.floor(v))
    c1, r1 = min(c0 + 1, 2048), min(r0 + 1, 2048)
    s, t = u - c0, v - r0
    h00 = hmap[r0, c0] * 0.01
    h10 = hmap[r0, c1] * 0.01
    h01 = hmap[r1, c0] * 0.01
    h11 = hmap[r1, c1] * 0.01
    return float((1.0 - s) * (1.0 - t) * h00 + s * (1.0 - t) * h10 + (1.0 - s) * t * h01 + s * t * h11)

con_path = '/home/ator/BF2/levels/slovenia_golovec/StaticObjects.con'
